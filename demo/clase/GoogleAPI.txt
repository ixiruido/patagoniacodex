function doGet(e) {
  // 1. Capturamos y normalizamos el email enviado por la web
  var emailUsuario = e.parameter.email ? e.parameter.email.toLowerCase().trim() : "";
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 2. Buscamos la hoja específica llamada "Clases"
  var hoja = ss.getSheetByName("Clases");
  var resultado = { clases: [] };

  // Error si la web no envía un email
  if (emailUsuario === "") {
    return ContentService.createTextOutput(JSON.stringify({ error: "Email requerido" }))
                         .setMimeType(ContentService.MimeType.JSON);
  }

  // Error si el docente le cambió el nombre a la pestaña sin querer
  if (!hoja) {
    return ContentService.createTextOutput(JSON.stringify({ error: "Error interno: No se encontró la pestaña 'Clases'" }))
                         .setMimeType(ContentService.MimeType.JSON);
  }

  var ultimaColumna = hoja.getLastColumn();
  var ultimaFila = hoja.getLastRow();

  // 3. Obtenemos la lista completa de emails de la columna A (desde la fila 7 hacia abajo)
  var listaEmails = [];
  if (ultimaFila >= 7) {
    listaEmails = hoja.getRange(7, 1, ultimaFila - 6, 1).getValues().map(function(r) {
      return r.toString().toLowerCase().trim();
    });
  }

  // Buscamos en qué posición de nuestro array está el alumno
  var filaAlumno = listaEmails.indexOf(emailUsuario);
  
  // 4. Recorremos horizontalmente cada columna (desde la columna 2 que es la B, hasta la última)
  for (var col = 2; col <= ultimaColumna; col++) {
    var nombreClase = hoja.getRange(1, col).getValue().toString().trim();
    if (nombreClase === "") break; // Si encuentra una columna vacía, se detiene

    var autorizado = false;

    // Si el alumno está registrado en la lista, verificamos si su celda dice exactamente "SI"
    if (filaAlumno !== -1) {
      var filaReal = filaAlumno + 7; // Ajustamos el índice a la fila real del Sheets
      var estadoAutorizacion = hoja.getRange(filaReal, col).getValue().toString().toUpperCase().trim();
      if (estadoAutorizacion === "SI") {
        autorizado = true;
      }
    }

    // Estructura de datos blindada: si no pagó, los enlaces viajan vacíos (null) hacia la web
    var datosClase = {
      nombre: nombreClase,
      acceso: autorizado,
      video: autorizado ? hoja.getRange(2, col).getValue().toString().trim() : null,
      pdf: autorizado ? hoja.getRange(3, col).getValue().toString().trim() : null,
      comentario: autorizado ? hoja.getRange(4, col).getValue().toString().trim() : null
    };

    resultado.clases.push(datosClase);
  }

  // Devolvemos la respuesta en formato JSON puro
  return ContentService.createTextOutput(JSON.stringify(resultado))
                       .setMimeType(ContentService.MimeType.JSON);
}

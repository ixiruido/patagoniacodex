function doGet() {

  const ID_PLANILLA =
    "1oQC-N91mh9gZdqTkinFRwrKQN7OIXTKEogYDKB5RRA0";

  const planilla =
    SpreadsheetApp.openById(ID_PLANILLA);

  const hoja =
    planilla.getSheetByName("Propiedades");

  const rango =
    hoja.getDataRange();

  const datos =
    rango.getValues();

  datos.shift();


  // ============================
  // CONFIGURACIÓN
  // ============================

  const whatsapp = "5492996100676";

  const mensajeWhatsapp = "Hola! Estoy interesado en esta propiedad: ";


  // ============================
  // Propiedades
  // ============================

  const propiedades =
    datos.map((fila, indice) => {

      let foto = "";


      const celdaFoto =
        rango
          .getCell(indice + 2, 7)
          .getValue();


      // Imagen insertada dentro de la celda

      if (
        celdaFoto &&
        typeof celdaFoto === "object" &&
        typeof celdaFoto.getContentUrl === "function"
      ) {

        foto =
          celdaFoto.getContentUrl();

      }


      // Limpiar URL

      if (
        typeof foto === "string"
      ) {

        const match =
          foto.match(
            /\((https?:\/\/[^)]+)\)/
          );


        if (match) {
          foto = match[1];
        }


        const inicio =
          foto.indexOf("https://");


        if (inicio > 0) {
          foto =
            foto.substring(inicio);
        }

      }


      return {

        tipo:
          fila[0],

        localidad:
          fila[1],

        direccion:
          fila[2],

        categoria:
          fila[3],

        tamanio:
          fila[4],

        precio:
          fila[5],

        foto:
          foto,

        detalles:
          fila[7],

        capacidad:
          fila[8],

        orden:
          fila[9],

        visible:
          fila[10],
  
        destacado:
          fila[11],
    
        stock:
          fila[12],
      };

    });


  // ============================
  // RESPUESTA
  // ============================

  const respuesta = {

    whatsapp:
      whatsapp,

    mensajeWhatsapp:
      mensajeWhatsapp,

    propiedades:
      propiedades

  };


  return ContentService
    .createTextOutput(
      JSON.stringify(respuesta)
    )
    .setMimeType(
      ContentService.MimeType.JSON
    );
}

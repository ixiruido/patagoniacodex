function doGet(e) {

  // ==========================================
  // CONFIGURACIÓN
  // ==========================================

  const NOMBRE_HOJA = "Clases";
  const FILA_ALUMNOS = 7;
  const COLUMNA_EMAIL = 1; // Columna A


  // ==========================================
  // OBTENER EMAIL
  // ==========================================

  const emailUsuario = e.parameter.email
    ? e.parameter.email.toLowerCase().trim()
    : "";


  // ==========================================
  // VALIDAR EMAIL
  // ==========================================

  if (emailUsuario === "") {

    return responderJSON({
      error: "Email requerido"
    });

  }


  // ==========================================
  // OBTENER GOOGLE SHEET
  // ==========================================

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const hoja = ss.getSheetByName(NOMBRE_HOJA);


  if (!hoja) {

    return responderJSON({
      error: "No se encontró la pestaña 'Clases'"
    });

  }


  // ==========================================
  // DATOS GENERALES
  // ==========================================

  const ultimaFila = hoja.getLastRow();
  const ultimaColumna = hoja.getLastColumn();

  const resultado = {
    email: emailUsuario,
    clases: []
  };


  // ==========================================
  // BUSCAR ALUMNO
  // ==========================================

  let filaAlumno = -1;

  if (ultimaFila >= FILA_ALUMNOS) {

    const emails = hoja
      .getRange(
        FILA_ALUMNOS,
        COLUMNA_EMAIL,
        ultimaFila - FILA_ALUMNOS + 1,
        1
      )
      .getValues();

    for (let i = 0; i < emails.length; i++) {

      const email = String(emails[i][0])
        .toLowerCase()
        .trim();

      if (email === emailUsuario) {

        filaAlumno = FILA_ALUMNOS + i;

        break;

      }

    }

  }


  // ==========================================
  // SI NO EXISTE EL EMAIL
  // ==========================================

  if (filaAlumno === -1) {

    return responderJSON({
      error: "El email no está registrado"
    });

  }


  // ==========================================
  // RECORRER TODAS LAS CLASES
  // ==========================================

  for (let col = 2; col <= ultimaColumna; col++) {

    const nombreClase = String(
      hoja.getRange(1, col).getValue()
    ).trim();


    // Si encontramos una columna sin nombre,
    // dejamos de buscar clases.

    if (nombreClase === "") {
      break;
    }


    // ========================================
    // COMPROBAR AUTORIZACIÓN
    // ========================================

    const estado = String(
      hoja.getRange(filaAlumno, col).getValue()
    )
      .toUpperCase()
      .trim();

    const autorizado = estado === "SI";


    // ========================================
    // CREAR OBJETO DE CLASE
    // ========================================

    const clase = {

      nombre: nombreClase,

      acceso: autorizado,

      video: null,

      pdf: null,

      comentario: null

    };


    // ========================================
    // SOLO ENTREGAR CONTENIDO SI AUTORIZADO
    // ========================================

    if (autorizado) {

      clase.video = String(
        hoja.getRange(2, col).getValue()
      ).trim();

      clase.pdf = String(
        hoja.getRange(3, col).getValue()
      ).trim();

      clase.comentario = String(
        hoja.getRange(4, col).getValue()
      ).trim();

    }


    resultado.clases.push(clase);

  }


  // ==========================================
  // RESPONDER
  // ==========================================

  return responderJSON(resultado);

}



// ============================================
// FUNCIÓN PARA RESPONDER JSON
// ============================================

function responderJSON(objeto) {

  return ContentService
    .createTextOutput(
      JSON.stringify(objeto)
    )
    .setMimeType(
      ContentService.MimeType.JSON
    );

}

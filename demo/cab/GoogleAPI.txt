/**
 * ==========================================
 * CONFIGURACIÓN
 * ==========================================
 */

const CONFIG = {

  usuario:
    "ixiruido",

  repositorio:
    "patagoniacodex",

  workflow:
    "actualizar-cabanas.yml",

  rama:
    "main",

  nombreDemo:
    "cab"
};


/**
 * ==========================================
 * GENERAR JSON
 * ==========================================
 *
 * Lee TODAS las cabañas desde la fila 2.
 *
 * Cada fila representa una cabaña.
 */

function doGet() {

  var sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getActiveSheet();


  // ==========================================
  // LEER TODAS LAS FILAS
  // ==========================================

  var ultimaFila =
    sheet.getLastRow();


  // Si solamente existe el encabezado
  if (ultimaFila < 2) {

    var respuestaVacia = {

      whatsapp:
        "5492996100676",

      mensajeWhatsapp:
        "Hola! Estoy interesado en este alojamiento: ",

      cabanas:
        []

    };


    return ContentService

      .createTextOutput(
        JSON.stringify(
          respuestaVacia
        )
      )

      .setMimeType(
        ContentService.MimeType.JSON
      );
  }


  // Leer desde fila 2 hasta la última fila
  // Columnas A a J
  var datos =
    sheet
      .getRange(
        2,
        1,
        ultimaFila - 1,
        10
      )
      .getValues();


  // ==========================================
  // CONFIGURACIÓN WHATSAPP
  // ==========================================

  var whatsapp =
    "5492996100676";

  var mensajeWhatsapp =
    "Hola! Estoy interesado en este alojamiento: ";


  // ==========================================
  // CABAÑAS
  // ==========================================

  var cabanas =

    datos

      .map(function(fila, indice) {

        // ====================================
        // IGNORAR FILAS SIN NOMBRE
        // ====================================

        if (
          !fila[0] ||
          String(fila[0]).trim() === ""
        ) {

          return null;

        }


        // ====================================
        // DATOS
        // ====================================

        var cabanaNombre =
          String(fila[0]);

        var estado =
          String(fila[1]);

        var precioBase =
          String(fila[2]);

        var descuento =
          String(fila[3]);

        var aviso =
          String(fila[4]);

        var ultimaActualizacion =
          String(fila[5]);


        // ====================================
        // WHATSAPP
        // ====================================

        var mensajeCompleto =
          "Hola! Estoy interesado en este alojamiento: " +
          cabanaNombre +
          " " +
          precioBase;


        var mensajeCodificado =
          encodeURIComponent(
            mensajeCompleto
          );


        var linkWhatsapp =
          "https://wa.me/" +
          whatsapp +
          "?text=" +
          mensajeCodificado;


        // ====================================
        // CABAÑA
        // ====================================

        return {

          nombre:
            cabanaNombre,

          estado:
            estado,

          precioBase:
            precioBase,

          descuento:
            descuento,

          aviso:
            aviso,

          ultimaActualizacion:
            ultimaActualizacion,

          whatsappLink:
            linkWhatsapp

        };

      })

      // Eliminar filas vacías
      .filter(function(cabana) {

        return cabana !== null;

      });


  // ==========================================
  // RESPUESTA
  // ==========================================

  var respuesta = {

    whatsapp:
      whatsapp,

    mensajeWhatsapp:
      mensajeWhatsapp,

    cabanas:
      cabanas

  };


  // ==========================================
  // DEVOLVER JSON
  // ==========================================

  return ContentService

    .createTextOutput(
      JSON.stringify(
        respuesta
      )
    )

    .setMimeType(
      ContentService.MimeType.JSON
    );
}


/**
 * ==========================================
 * BOTÓN "GUARDAR CABAÑAS"
 * ==========================================
 */

function guardarCabanas() {

  var ui =
    SpreadsheetApp.getUi();


  // ------------------------------------------
  // CONFIRMACIÓN
  // ------------------------------------------

  var respuesta =
    ui.alert(

      "Guardar cambios",

      "¿Querés publicar los cambios de las cabañas en la web?",

      ui.ButtonSet.YES_NO

    );


  // Cancelar
  if (
    respuesta !== ui.Button.YES
  ) {

    return;

  }


  // ------------------------------------------
  // ENVIAR A GITHUB
  // ------------------------------------------

  try {

    dispararGitHub();


    // ----------------------------------------
    // ÉXITO
    // ----------------------------------------

    ui.alert(

      "Cambios enviados",

      "Los cambios fueron enviados correctamente a la Web.\n\n" +
      "En instantes se actualizarán los alquileres.",

      ui.ButtonSet.OK

    );


  } catch (error) {

    // ----------------------------------------
    // ERROR
    // ----------------------------------------

    ui.alert(

      "Error",

      "No se pudieron enviar los cambios a GitHub.\n\n" +
      error.message,

      ui.ButtonSet.OK

    );


    Logger.log(
      "ERROR: " +
      error.message
    );

  }
}


/**
 * ==========================================
 * ENVIAR WORKFLOW A GITHUB
 * ==========================================
 */

function dispararGitHub() {

  // ------------------------------------------
  // TOKEN
  // ------------------------------------------

  var token =
    PropertiesService
      .getScriptProperties()
      .getProperty(
        "GITHUB_TOKEN"
      );


  if (!token) {

    throw new Error(

      "No se encontró GITHUB_TOKEN " +
      "en las propiedades del proyecto."

    );

  }


  // ------------------------------------------
  // URL
  // ------------------------------------------

  var url =

    "https://api.github.com/repos/" +

    CONFIG.usuario +
    "/" +

    CONFIG.repositorio +

    "/actions/workflows/" +

    CONFIG.workflow +

    "/dispatches";


  // ------------------------------------------
  // PAYLOAD
  // ------------------------------------------

  var payload = {

    ref:
      CONFIG.rama,

    inputs: {

      demo_target:
        CONFIG.nombreDemo

    }

  };


  // ------------------------------------------
  // OPCIONES
  // ------------------------------------------

  var options = {

    method:
      "post",

    contentType:
      "application/json",

    payload:
      JSON.stringify(
        payload
      ),

    headers: {

      Authorization:
        "Bearer " + token,

      Accept:
        "application/vnd.github+json",

      "X-GitHub-Api-Version":
        "2022-11-28"

    },

    muteHttpExceptions:
      true

  };


  // ------------------------------------------
  // LLAMADA A GITHUB
  // ------------------------------------------

  var response =
    UrlFetchApp.fetch(
      url,
      options
    );


  var status =
    response.getResponseCode();

  var body =
    response.getContentText();


  // ------------------------------------------
  // LOG
  // ------------------------------------------

  Logger.log(
    "GitHub HTTP STATUS: " +
    status
  );

  Logger.log(
    "GitHub RESPONSE: " +
    body
  );


  // ------------------------------------------
  // COMPROBAR
  // ------------------------------------------

  if (
    status !== 204 &&
    status !== 200
  ) {

    throw new Error(

      "GitHub rechazó el workflow.\n\n" +

      "HTTP: " +
      status +

      "\n" +

      "Respuesta: " +
      body

    );

  }


  Logger.log(
    "Workflow enviado correctamente."
  );

  Logger.log(
    "Cabañas publicadas."
  );
}
